cmake_minimum_required(VERSION 3.22.1)
project(mobile_whisper LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimize on-device inference even in debug APKs" FORCE)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    add_library(mobile_whisper SHARED unsupported.cpp)
    return()
endif()
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
FetchContent_Declare(whisper URL https://github.com/ggml-org/whisper.cpp/archive/refs/tags/v1.8.3.tar.gz URL_HASH SHA256=870ba21409cdf66697dc4db15ebdb13bc67037d76c7cc63756c81471d8f1731a)
FetchContent_MakeAvailable(whisper)
add_library(mobile_whisper SHARED whisper_jni.cpp)
target_link_libraries(mobile_whisper PRIVATE whisper android log)
target_compile_features(mobile_whisper PRIVATE cxx_std_17)
target_link_options(mobile_whisper PRIVATE "-Wl,-z,max-page-size=16384")
